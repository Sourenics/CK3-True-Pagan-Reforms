# True Pagan Reforms - scripted effects.

# Faith scope (the new, reformed faith), from tpr_on_faith_created.
# Picks who shaped the reform, and if an AI did, swaps some of the reformed rite's tenets and doctrines for ones that fit them.
tpr_vary_reformed_faith_effect = {
	if = {
		limit = { exists = scope:founder }
		scope:founder = { save_scope_as = tpr_founder }
	}
	else_if = {
		limit = { exists = main_rite.head_of_rite }
		main_rite.head_of_rite = { save_scope_as = tpr_founder }
	}
	if = {
		limit = {
			exists = scope:tpr_founder
			scope:tpr_founder = { is_ai = yes }
		}
		main_rite = { save_scope_as = tpr_rite }
		scope:old_faith.main_rite = { save_scope_as = tpr_old_rite }
		tpr_vary_rite_effect = yes
		debug_log = "True Pagan Reforms: an AI reform was varied"
		debug_log_scopes = yes
	}
}

# Needs scope:tpr_rite and scope:tpr_founder; scope:tpr_old_rite is optional.
# Makes 1 or 2 tenet swaps, following the tenet game rule (a reformed rite has 3 tenets, so at least 1 always stays),
# or replaces all of them with No Kinship; then changes doctrines following the doctrine game rule.
tpr_vary_rite_effect = {
	# Only enforce the editor rules if the rite followed them to begin with, so an odd starting faith never blocks the mod.
	if = {
		limit = { tpr_rite_follows_editor_rules_trigger = yes }
		save_scope_value_as = { name = tpr_check_rules value = yes }
	}
	if = {
		limit = { has_game_rule = tpr_variation_no_kinship }
		# Every tenet goes, and none of the old faith's tenets can come back. Each swap takes out a tenet the reform
		# started with; a swap that finds nothing to add puts its tenet back, so up to 2 extra tries replace it.
		save_scope_value_as = { name = tpr_no_kinship value = yes }
		while = {
			count = 5
			if = {
				limit = {
					scope:tpr_rite = {
						any_rite_tenet = { tpr_can_remove_tenet_trigger = yes }
					}
				}
				tpr_swap_one_tenet_effect = yes
			}
		}
	}
	else_if = {
		limit = { has_game_rule = tpr_variation_strong }
		tpr_swap_one_tenet_effect = yes
		tpr_swap_one_tenet_effect = yes
	}
	else_if = {
		limit = { has_game_rule = tpr_variation_light }
		tpr_swap_one_tenet_effect = yes
	}
	else = {
		tpr_swap_one_tenet_effect = yes
		random = {
			chance = 50
			tpr_swap_one_tenet_effect = yes
		}
	}
	# Doctrines go after tenets, so a tenet just added (Carnal Exaltation) can open a doctrine (virtuous deviancy).
	tpr_vary_doctrines_effect = yes
}

# Needs scope:tpr_rite and scope:tpr_founder.
# Moves 0 to 6 doctrines one step, following the doctrine game rule. Each group changes at most once.
# Then sets up a head of faith title if the faith now needs one.
tpr_vary_doctrines_effect = {
	if = {
		limit = { has_game_rule = tpr_doctrine_variation_light }
		tpr_change_one_doctrine_effect = yes
	}
	else_if = {
		limit = { has_game_rule = tpr_doctrine_variation_strong }
		tpr_change_one_doctrine_effect = yes
		tpr_change_one_doctrine_effect = yes
		tpr_change_one_doctrine_effect = yes
		random = {
			chance = 50
			tpr_change_one_doctrine_effect = yes
		}
	}
	else_if = {
		limit = { has_game_rule = tpr_doctrine_variation_radical }
		# 4, 5 or 6, even odds.
		tpr_change_one_doctrine_effect = yes
		tpr_change_one_doctrine_effect = yes
		tpr_change_one_doctrine_effect = yes
		tpr_change_one_doctrine_effect = yes
		random_list = {
			1 = {}
			1 = {
				tpr_change_one_doctrine_effect = yes
			}
			1 = {
				tpr_change_one_doctrine_effect = yes
				tpr_change_one_doctrine_effect = yes
			}
		}
	}
	# Moderate, also when the rule did not exist yet in the save.
	else_if = {
		limit = { NOT = { has_game_rule = tpr_doctrine_variation_none } }
		tpr_change_one_doctrine_effect = yes
		tpr_change_one_doctrine_effect = yes
	}
	tpr_set_up_head_of_faith_effect = yes
}

# Needs scope:tpr_rite. Lets the game sort out the head of faith title after the doctrines changed, with the same
# events the game fires after a rite is edited (on_rite_updated). Each one checks its own conditions, so they only
# act when needed: create or restore a title for a spiritual or temporal head (if the game's requirements are met;
# otherwise the faith can create it later with the game's decision), switch its succession law, or destroy a title
# the faith no longer has a head for.
tpr_set_up_head_of_faith_effect = {
	if = {
		limit = {
			OR = {
				scope:tpr_doctrine_1 ?= { tpr_is_head_of_faith_doctrine_trigger = yes }
				scope:tpr_doctrine_2 ?= { tpr_is_head_of_faith_doctrine_trigger = yes }
				scope:tpr_doctrine_3 ?= { tpr_is_head_of_faith_doctrine_trigger = yes }
				scope:tpr_doctrine_4 ?= { tpr_is_head_of_faith_doctrine_trigger = yes }
				scope:tpr_doctrine_5 ?= { tpr_is_head_of_faith_doctrine_trigger = yes }
				scope:tpr_doctrine_6 ?= { tpr_is_head_of_faith_doctrine_trigger = yes }
			}
		}
		scope:tpr_rite = { save_scope_as = rite }
		scope:tpr_rite.head_of_rite ?= {
			trigger_event = faith_creation.0004 # Update head of faith succession law
			trigger_event = faith_creation.0001 # Temporal head of faith setup
			trigger_event = faith_creation.0002 # Spiritual head of faith setup
			trigger_event = faith_creation.0011 # Restore temporal head of faith
			trigger_event = faith_creation.0012 # Restore spiritual head of faith
			trigger_event = faith_creation.1100 # Destroy the title if the faith no longer has a head
		}
	}
}

# Needs scope:tpr_rite and scope:tpr_founder.
# Moves one doctrine to a valid next option, preferring those that fit the founder. A change that breaks the faith
# editor's rules is undone and another doctrine is tried, up to 3 times. Does nothing if no doctrine qualifies.
tpr_change_one_doctrine_effect = {
	clear_saved_scope = tpr_doctrine_to_add
	clear_saved_scope = tpr_no_more_doctrines
	while = {
		count = 3
		if = {
			limit = {
				NOT = { exists = scope:tpr_doctrine_to_add }
				NOT = { exists = scope:tpr_no_more_doctrines }
			}
			tpr_try_doctrine_effect = yes
		}
	}
	if = {
		limit = { exists = scope:tpr_doctrine_to_add }
		# Remember it, so the same group is not changed twice.
		if = {
			limit = { NOT = { exists = scope:tpr_doctrine_1 } }
			scope:tpr_doctrine_to_add = { save_scope_as = tpr_doctrine_1 }
		}
		else_if = {
			limit = { NOT = { exists = scope:tpr_doctrine_2 } }
			scope:tpr_doctrine_to_add = { save_scope_as = tpr_doctrine_2 }
		}
		else_if = {
			limit = { NOT = { exists = scope:tpr_doctrine_3 } }
			scope:tpr_doctrine_to_add = { save_scope_as = tpr_doctrine_3 }
		}
		else_if = {
			limit = { NOT = { exists = scope:tpr_doctrine_4 } }
			scope:tpr_doctrine_to_add = { save_scope_as = tpr_doctrine_4 }
		}
		else_if = {
			limit = { NOT = { exists = scope:tpr_doctrine_5 } }
			scope:tpr_doctrine_to_add = { save_scope_as = tpr_doctrine_5 }
		}
		else = {
			scope:tpr_doctrine_to_add = { save_scope_as = tpr_doctrine_6 }
		}
	}
}

# One attempt of tpr_change_one_doctrine_effect.
tpr_try_doctrine_effect = {
	clear_saved_scope = tpr_doctrine_candidate
	clear_saved_scope = tpr_doctrine_previous
	random_doctrine = {
		limit = {
			NOT = { is_in_list = tpr_rejected_doctrines }
			tpr_can_change_to_doctrine_trigger = yes
		}
		weight = {
			base = 10
			modifier = {
				add = tpr_doctrine_affinity_value
			}
		}
		save_scope_as = tpr_doctrine_candidate
	}
	if = {
		limit = { exists = scope:tpr_doctrine_candidate }
		random_doctrine = {
			limit = {
				tpr_doctrine_same_group_as_candidate_trigger = yes
				save_temporary_scope_as = tpr_doctrine_check
				scope:tpr_rite = { rite_has_doctrine = scope:tpr_doctrine_check }
			}
			save_scope_as = tpr_doctrine_previous
		}
		scope:tpr_rite = { change_rite_doctrine = scope:tpr_doctrine_candidate }
		if = {
			limit = {
				exists = scope:tpr_check_rules
				NOT = { tpr_rite_follows_editor_rules_trigger = yes }
			}
			if = {
				limit = { exists = scope:tpr_doctrine_previous }
				scope:tpr_rite = { change_rite_doctrine = scope:tpr_doctrine_previous }
			}
			scope:tpr_doctrine_candidate = { add_to_temporary_list = tpr_rejected_doctrines }
		}
		else = {
			scope:tpr_doctrine_candidate = { save_scope_as = tpr_doctrine_to_add }
		}
	}
	else = {
		save_scope_value_as = { name = tpr_no_more_doctrines value = yes }
	}
}

# Needs scope:tpr_rite and scope:tpr_founder; scope:tpr_old_rite is optional.
# Removes one tenet and adds one that fits the founder. A tenet that breaks the faith editor's rules is taken out
# again and another is tried, up to 3 times. If nothing can be added, the removed tenet comes back.
# With No Kinship, Carnal Exaltation can go too: a Virtuous deviancy then moves back to Accepted, as it needs that tenet.
tpr_swap_one_tenet_effect = {
	clear_saved_scope = tpr_tenet_to_remove
	clear_saved_scope = tpr_tenet_to_add
	clear_saved_scope = tpr_no_more_tenets
	scope:tpr_rite = {
		random_rite_tenet = {
			limit = { tpr_can_remove_tenet_trigger = yes }
			weight = {
				base = 100
				# A tenet that already fits the founder is more likely to stay.
				modifier = {
					factor = 0.2
					tpr_tenet_affinity_value > 0
				}
				# So is one inherited from the old faith.
				modifier = {
					factor = 0.5
					tpr_tenet_is_inherited_trigger = yes
				}
			}
			save_scope_as = tpr_tenet_to_remove
		}
	}
	if = {
		limit = { exists = scope:tpr_tenet_to_remove }
		scope:tpr_rite = { remove_rite_tenet = scope:tpr_tenet_to_remove }
		clear_saved_scope = tpr_deviancy_lowered
		if = {
			limit = {
				scope:tpr_tenet_to_remove = tenet:tenet_carnal_exaltation
				scope:tpr_rite = { rite_has_doctrine = doctrine_deviancy_virtuous }
			}
			scope:tpr_rite = { change_rite_doctrine = doctrine:doctrine_deviancy_accepted }
			save_scope_value_as = { name = tpr_deviancy_lowered value = yes }
		}
		while = {
			count = 3
			if = {
				limit = {
					NOT = { exists = scope:tpr_tenet_to_add }
					NOT = { exists = scope:tpr_no_more_tenets }
				}
				tpr_try_tenet_effect = yes
			}
		}
		if = {
			limit = { exists = scope:tpr_tenet_to_add }
			# Remember it, so the second swap never takes it back out.
			if = {
				limit = { NOT = { exists = scope:tpr_added_1 } }
				scope:tpr_tenet_to_add = { save_scope_as = tpr_added_1 }
			}
			else_if = {
				limit = { NOT = { exists = scope:tpr_added_2 } }
				scope:tpr_tenet_to_add = { save_scope_as = tpr_added_2 }
			}
			else = {
				scope:tpr_tenet_to_add = { save_scope_as = tpr_added_3 }
			}
		}
		else = {
			scope:tpr_rite = { add_rite_tenet = scope:tpr_tenet_to_remove }
			if = {
				limit = { exists = scope:tpr_deviancy_lowered }
				scope:tpr_rite = { change_rite_doctrine = doctrine:doctrine_deviancy_virtuous }
			}
		}
	}
}

# One attempt of tpr_swap_one_tenet_effect.
tpr_try_tenet_effect = {
	clear_saved_scope = tpr_tenet_candidate
	random_tenet = {
		limit = {
			NOT = { this = scope:tpr_tenet_to_remove }
			NOT = { is_in_list = tpr_rejected_tenets }
			tpr_can_add_tenet_trigger = yes
		}
		weight = {
			base = 10
			modifier = {
				add = tpr_tenet_affinity_value
			}
			# Syncretism: no base chance, only the weight of the founder's bonds.
			modifier = {
				tpr_is_syncretism_tenet_trigger = yes
				add = -10
			}
			modifier = {
				add = tpr_syncretism_affinity_value
			}
		}
		save_scope_as = tpr_tenet_candidate
	}
	if = {
		limit = { exists = scope:tpr_tenet_candidate }
		scope:tpr_rite = { add_rite_tenet = scope:tpr_tenet_candidate }
		if = {
			limit = {
				exists = scope:tpr_check_rules
				NOT = { tpr_rite_follows_editor_rules_trigger = yes }
			}
			scope:tpr_rite = { remove_rite_tenet = scope:tpr_tenet_candidate }
			scope:tpr_tenet_candidate = { add_to_temporary_list = tpr_rejected_tenets }
		}
		else = {
			scope:tpr_tenet_candidate = { save_scope_as = tpr_tenet_to_add }
		}
	}
	else = {
		save_scope_value_as = { name = tpr_no_more_tenets value = yes }
	}
}
