# True Pagan Reforms - scripted triggers.

# Faith scope (the new faith), from on_faith_created.
# A reformation: the old faith is unreformed, and the new faith did not come from a rite diverging too far.
tpr_is_pagan_reformation_trigger = {
	exists = scope:old_faith
	scope:old_faith = { has_doctrine_parameter = unreformed }
	NOT = { scope:diverged ?= yes }
	exists = main_rite
}

# Tenet scope. Needs scope:tpr_rite. Quick pre-check of combinations that contradict each other, so obviously
# clashing tenets are not drawn at all. The full set of editor rules is checked after adding (tpr_editor_rules_triggers.txt).
tpr_tenet_has_no_conflict_trigger = {
	trigger_if = {
		limit = {
			OR = {
				this = tenet:tenet_warmonger
				this = tenet:tenet_human_sacrifice
				this = tenet:tenet_gruesome_festivals
				this = tenet:tenet_ritual_cannibalism
				this = tenet:tenet_sacrificial_ceremonies
			}
		}
		scope:tpr_rite = {
			NOR = {
				rite_has_tenet = tenet_pacifism
				rite_has_tenet = tenet_dharmic_pacifism
			}
		}
	}
	trigger_if = {
		limit = {
			OR = {
				this = tenet:tenet_hedonistic
				this = tenet:tenet_carnal_exaltation
			}
		}
		scope:tpr_rite = {
			NOT = { rite_has_tenet = tenet_asceticism }
		}
	}
	# The faith editor only lets Adaptability go with pluralism.
	trigger_if = {
		limit = { this = tenet:tenet_adaptive }
		scope:tpr_rite = { rite_has_doctrine = doctrine_pluralism_pluralistic }
	}
	trigger_if = {
		limit = { this = tenet:tenet_asceticism }
		scope:tpr_rite = {
			NOR = {
				rite_has_tenet = tenet_hedonistic
				rite_has_tenet = tenet_carnal_exaltation
			}
		}
	}
}

# Tenet scope. scope:tpr_old_rite is optional. The unreformed rite also has this tenet.
tpr_tenet_is_inherited_trigger = {
	save_temporary_scope_as = tpr_inherit_check
	scope:tpr_old_rite ?= { rite_has_tenet = scope:tpr_inherit_check }
}

# Tenet scope (a tenet of scope:tpr_rite). The tenet can be swapped out:
# - never a tenet this mod just added;
# - with No Kinship (scope:tpr_no_kinship), any other tenet, so all of them are replaced;
# - otherwise any tenet except syncretism, which only comes from the founder's bonds, and never the last tenet
#   inherited from the old faith, so the reform keeps its kinship with it.
tpr_can_remove_tenet_trigger = {
	NOR = {
		scope:tpr_added_1 ?= this
		scope:tpr_added_2 ?= this
		scope:tpr_added_3 ?= this
	}
	trigger_if = {
		limit = { NOT = { exists = scope:tpr_no_kinship } }
		NOT = { tpr_is_syncretism_tenet_trigger = yes }
		# Virtuous deviancy needs Carnal Exaltation.
		NAND = {
			this = tenet:tenet_carnal_exaltation
			scope:tpr_rite = { rite_has_doctrine = doctrine_deviancy_virtuous }
		}
		trigger_if = {
			limit = { tpr_tenet_is_inherited_trigger = yes }
			scope:tpr_rite = {
				any_rite_tenet = {
					count >= 2
					tpr_tenet_is_inherited_trigger = yes
				}
			}
		}
	}
}

# Tenet scope. Needs scope:tpr_rite. The tenet can be added to the rite: any tenet the game lets the rite choose
# (rite_can_choose_tenet checks religion, DLC and character requirements). Syncretism tenets only qualify through a
# bond of the founder (see tpr_syncretism_triggers.txt). With No Kinship, no tenet of the old faith qualifies.
# The editor's compatibility rules are checked after adding.
tpr_can_add_tenet_trigger = {
	trigger_if = {
		limit = { exists = scope:tpr_no_kinship }
		NOT = { tpr_tenet_is_inherited_trigger = yes }
	}
	OR = {
		NOT = { tpr_is_syncretism_tenet_trigger = yes }
		AND = {
			tpr_is_syncretism_tenet_trigger = yes
			tpr_syncretism_affinity_value > 0
			tpr_syncretism_is_allowed_trigger = yes
		}
	}
	save_temporary_scope_as = tpr_tenet_check
	scope:tpr_rite = {
		NOT = { rite_has_tenet = scope:tpr_tenet_check }
		rite_can_choose_tenet = scope:tpr_tenet_check
	}
	tpr_tenet_has_no_conflict_trigger = yes
}
