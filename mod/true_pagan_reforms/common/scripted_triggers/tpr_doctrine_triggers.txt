# Doctrine scope. Needs scope:tpr_rite. Conditions the faith editor checks for these doctrines,
# written out here because the game only checks some of them while a faith is being edited.
# Not offered at all: the Hajj (Islam only) and conditional clerical succession (cannot be picked anew).
tpr_doctrine_is_allowed_trigger = {
	trigger_if = {
		limit = { this = doctrine:doctrine_gender_male_dominated }
		NOT = { has_game_rule = full_gender_equality }
		NAND = {
			exists = scope:tpr_rite.faith.religious_head
			scope:tpr_rite.faith.religious_head.culture = { has_cultural_parameter = female_only_inheritance }
		}
	}
	trigger_if = {
		limit = { this = doctrine:doctrine_gender_female_dominated }
		NOT = { has_game_rule = full_gender_equality }
		NAND = {
			exists = scope:tpr_rite.faith.religious_head
			scope:tpr_rite.faith.religious_head.culture = { has_cultural_parameter = male_only_inheritance }
		}
	}
	trigger_if = {
		limit = { this = doctrine:doctrine_pluralism_pluralistic }
		scope:tpr_rite = { NOT = { rite_has_tenet = tenet_pentarchy } }
	}
	trigger_if = {
		limit = {
			OR = {
				this = doctrine:doctrine_theocracy_temporal
				this = doctrine:doctrine_clerical_succession_spiritual_appointment
				this = doctrine:doctrine_clerical_succession_spiritual_fixed_appointment
			}
		}
		scope:tpr_rite = { NOT = { rite_has_doctrine = doctrine_temporal_head } }
	}
	trigger_if = {
		limit = { this = doctrine:doctrine_pilgrimage_mandatory }
		NOT = { scope:tpr_rite.faith = { religion_tag = islam_religion } }
	}
	# Adaptability needs pluralism.
	trigger_if = {
		limit = {
			OR = {
				this = doctrine:doctrine_pluralism_fundamentalist
				this = doctrine:doctrine_pluralism_righteous
			}
		}
		scope:tpr_rite = { NOT = { rite_has_tenet = tenet_adaptive } }
	}
	# Deviancy can only become a virtue in a faith that exalts the flesh.
	trigger_if = {
		limit = { this = doctrine:doctrine_deviancy_virtuous }
		scope:tpr_rite = { rite_has_tenet = tenet_carnal_exaltation }
	}
}
